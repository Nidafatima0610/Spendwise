/// Calculates spending share percentage with zero-division safety
double _calculatePercentage(int categorySpentMinor, int totalMonthlyExpenseMinor) {
  // Zero-division safeguard for months with no recorded expenses
  if (totalMonthlyExpenseMinor <= 0) return 0.0;
  final double percentage = (categorySpentMinor / totalMonthlyExpenseMinor) * 100.0;
  return percentage.clamp(0.0, 100.0);
}

/// Renders proportional animated horizontal spending bars
Widget _buildCategoryBar(CategoryModel category, int spentAmountMinor, double sharePercentage) {
  return Padding(
    padding: const EdgeInsets.only(bottom: AppDimensions.space16),
    child: Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Row(
          mainAxisAlignment: MainAxisAlignment.spaceBetween,
          children: [
            Text(category.name, style: AppTypography.bodyMedium),
            Text('${sharePercentage.toStringAsFixed(1)}% • ${CurrencyFormatter.formatMinorUnits(spentAmountMinor)}'),
          ],
        ),
        const SizedBox(height: AppDimensions.space8),
        ClipRRect(
          borderRadius: BorderRadius.circular(AppDimensions.radiusFull),
          child: LinearProgressIndicator(
            value: sharePercentage / 100.0,
            valueColor: AlwaysStoppedAnimation<Color>(Color(category.colorValue)),
            backgroundColor: AppColors.surfaceLavender,
            minHeight: 10,
          ),
        ),
      ],
    ),
  );
}
