/// Calculates spending share percentage with zero-division safeguard
double _calculatePercentage(int categorySpentMinor, int totalMonthlyExpenseMinor) {
  // Zero-division safeguard for months with no recorded expenses
  if (totalMonthlyExpenseMinor <= 0) return 0.0;
  final double percentage = (categorySpentMinor / totalMonthlyExpenseMinor) * 100.0;
  return percentage.clamp(0.0, 100.0);
}

/// Multi-axis filtering query combining active month, category chip, and search
List<TransactionModel> get filteredTransactions {
  return monthTransactions.where((tx) {
    // 1. Category Filter
    if (_selectedCategoryId != null && tx.categoryId != _selectedCategoryId) {
      return false;
    }
    // 2. Transaction Type Filter (Income / Expense / All)
    if (_selectedType == 'Expenses' && !tx.isExpense) return false;
    if (_selectedType == 'Income' && !tx.isIncome) return false;
    return true;
  }).toList();
}
