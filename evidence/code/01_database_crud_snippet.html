/// Inserts a new transaction into SQLite using parameterized ConflictAlgorithm.replace
Future<int> insertTransaction(TransactionModel transaction) async {
  final activeDb = await db;
  return await activeDb.insert(DatabaseConstants.tableTransactions, transaction.toMap(), conflictAlgorithm: ConflictAlgorithm.replace);
}

/// Retrieves all transactions ordered newest first (DESC)
Future<List<TransactionModel>> getAllTransactions() async {
  final activeDb = await db;
  final maps = await activeDb.query(DatabaseConstants.tableTransactions, orderBy: '${DatabaseConstants.colTxDate} DESC');
  return maps.map((map) => TransactionModel.fromMap(map)).toList();
}

/// Updates an existing transaction in SQLite using parameterized ID match
Future<int> updateTransaction(TransactionModel transaction) async {
  final activeDb = await db;
  return await activeDb.update(DatabaseConstants.tableTransactions, transaction.toMap(),
      where: '${DatabaseConstants.colTxId} = ?', whereArgs: [transaction.id]);
}

/// Permanently deletes a transaction by ID
Future<int> deleteTransaction(String id) async {
  final activeDb = await db;
  return await activeDb.delete(DatabaseConstants.tableTransactions,
      where: '${DatabaseConstants.colTxId} = ?', whereArgs: [id]);
}

/// Calculates net total balance in integer minor units (paisas)
Future<int> getTotalBalance() async {
  final activeDb = await db;
  final inc = await activeDb.rawQuery('SELECT COALESCE(SUM(amount), 0) as t FROM transactions WHERE type = ?', ['income']);
  final exp = await activeDb.rawQuery('SELECT COALESCE(SUM(amount), 0) as t FROM transactions WHERE type = ?', ['expense']);
  return ((inc.first['t'] as int?) ?? 0) - ((exp.first['t'] as int?) ?? 0);
}
