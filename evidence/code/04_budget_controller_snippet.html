/// Internal worker to query SQLite for budgets and calculate metrics
Future<void> _loadBudgetsAndMetrics() async {
  // 1. Retrieve all category budgets with joined spent amounts for active month
  _budgets = await _dbService.getBudgetsByMonth(_selectedMonthYear);

  int totalCap = 0;
  int totalSpent = 0;
  int overCount = 0;

  // 2. Calculate totals and check warning / over-budget conditions
  for (final b in _budgets) {
    totalCap += b.limitAmount;
    totalSpent += b.spentAmount;
    if (b.isOverBudget) {
      overCount++; // Trigger over-budget alert status
    }
  }

  _totalBudgetCapMinor = totalCap;
  _totalBudgetSpentMinor = totalSpent;
  _overBudgetCount = overCount;
}

/// Evaluates threshold indicators on CategoryBudgetCard
/// - On Track: spent < 80% of limit
/// - Warning / Near Limit: 80% <= spent < 100% of limit
/// - Over Budget: spent >= 100% of limit
bool get isWarning => spentPercentage >= 80.0 && spentPercentage < 100.0;
bool get isOverBudget => spentAmount >= limitAmount;
double get spentPercentage => limitAmount > 0 ? (spentAmount / limitAmount) * 100 : 0.0;
