Future<void> _submitForm() async {
  FocusScope.of(context).unfocus();

  // 1. Prevent duplicate submissions
  if (_isSubmitting) return;

  // 2. Validate Form Key (empty check & format)
  if (!_formKey.currentState!.validate()) return;

  // 3. Validate Category Selection
  if (_selectedCategory == null) {
    _showSnackBar('Please select a category for this transaction.');
    return;
  }

  // 4. Validate Amount range & integer minor unit parsing (paisas)
  final int minorUnits = CurrencyFormatter.parseToMinorUnits(_amountController.text);
  if (minorUnits <= 0) {
    _showSnackBar('Amount must be greater than zero.');
    return;
  }
  if (minorUnits > 10000000000) { // 100 Million PKR ceiling
    _showSnackBar('Amount exceeds the maximum supported limit.');
    return;
  }

  setState(() => _isSubmitting = true);

  try {
    // Inserts new transaction or updates existing record without duplicating
    final success = widget.isEditing
        ? await _controller.updateTransaction(updatedTx)
        : await _controller.addTransaction(newTx);
    ...
  } finally {
    if (mounted) setState(() => _isSubmitting = false);
  }
}
