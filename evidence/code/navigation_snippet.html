class AppScaffold extends StatefulWidget { ... }

class _AppScaffoldState extends State<AppScaffold> {
  int _currentIndex = 0;

  @override
  Widget build(BuildContext context) {
    final screens = [
      DashboardScreen(
        onNavigateToHistory: () => _onTabSelected(1),
        onNavigateToAddTransaction: () => _onTabSelected(2),
        onNavigateToBudgets: () => _onTabSelected(3),
      ),
      HistoryScreen(onAddTransactionTap: () => _onTabSelected(2)),
      AddTransactionScreen(onTransactionSavedPreview: () => _onTabSelected(0)),
      const BudgetsScreen(),
      const SettingsScreen(),
    ];

    return Scaffold(
      // Preserves scroll offset & form state across all 5 destinations
      body: IndexedStack(index: _currentIndex, children: screens),
      bottomNavigationBar: _buildBottomNavigationBar(),
    );
  }
}
