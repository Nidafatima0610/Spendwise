class TransactionController extends ChangeNotifier {
  static TransactionController instance = TransactionController();

  /// Adds a new transaction to SQLite and triggers real-time UI rebuild
  Future<bool> addTransaction(TransactionModel transaction) async {
    try {
      final rows = await _dbService.insertTransaction(transaction);
      if (rows > 0) {
        await _loadTransactionsAndMetrics();
        notifyListeners(); // Updates DashboardScreen and HistoryScreen via ListenableBuilder
        return true;
      }
      return false;
    } catch (e) {
      return false;
    }
  }

  /// Updates an existing transaction in SQLite and recalculates metrics
  Future<bool> updateTransaction(TransactionModel transaction) async {
    try {
      final rows = await _dbService.updateTransaction(transaction);
      if (rows > 0) {
        await _loadTransactionsAndMetrics();
        notifyListeners();
        return true;
      }
      return false;
    } catch (e) {
      return false;
    }
  }

  /// Deletes transaction from SQLite by ID and refreshes state
  Future<bool> deleteTransaction(String id) async {
    try {
      final rows = await _dbService.deleteTransaction(id);
      if (rows > 0) {
        await _loadTransactionsAndMetrics();
        notifyListeners();
        return true;
      }
      return false;
    } catch (e) {
      return false;
    }
  }
}
