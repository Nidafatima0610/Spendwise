enum TransactionType { income, expense; }

/// Core transaction data model representing income or expense entries.
/// NOTE: [amount] is stored strictly as an INTEGER in minor currency units (paisas)
/// to prevent IEEE 754 floating-point rounding errors. 1 PKR = 100 Paisa.
class TransactionModel {
  final String id;
  final TransactionType type;
  final int amount; // Integer in minor units (e.g., 250000 = PKR 2,500.00)
  final DateTime date;
  final String categoryId;
  final String? note;

  const TransactionModel({
    required this.id,
    required this.type,
    required this.amount,
    required this.date,
    required this.categoryId,
    this.note,
  });

  double get amountInMajorUnits => amount / 100.0;
  String get formattedAmount => CurrencyFormatter.formatMinorUnits(amount);
  bool get isIncome => type == TransactionType.income;
  bool get isExpense => type == TransactionType.expense;

  Map<String, dynamic> toMap() => {
    'id': id,
    'type': type.name,
    'amount': amount,
    'date': date.toIso8601String(),
    'category_id': categoryId,
    'note': note,
  };
}
